local function SaveButtonPosition(pos, name)
    pcall(function()
        if writefile then
            local folder = 'qqwmnt'
            if not isfolder(folder) then
                makefolder(folder)
            end
            writefile(folder .. '/' .. name .. 'Pos.json', HttpService:JSONEncode({
                X = { Scale = pos.X.Scale, Offset = pos.X.Offset },
                Y = { Scale = pos.Y.Scale, Offset = pos.Y.Offset },
            }))
        end
    end)
end

local function LoadButtonPosition(name)
    local success, result = pcall(function()
        if readfile and isfile and isfile('qqwmnt/' .. name .. 'Pos.json') then
            return HttpService:JSONDecode(readfile('qqwmnt/' .. name .. 'Pos.json'))
        end
    end)
    if success and result then
        return UDim2.new(result.X.Scale, result.X.Offset, result.Y.Scale, result.Y.Offset)
    end
    return nil
end

    label.TextScaled = false
    label.TextSize = 13

    label.TextColor3 = Color3.fromRGB(255, 255, 255)

        size = newSize

    local function EnsureBombInBackpack()
        local character = LocalPlayer.Character
        local backpack = LocalPlayer:FindFirstChild('Backpack')
        if not character or not backpack then return nil end

        local bomb = character:FindFirstChild(config.BombName) or backpack:FindFirstChild(config.BombName)
        if not bomb then

                ReplicatedStorage.Remotes.Extras.ReplicateToy:InvokeServer(config.BombName)

            task.wait(0.15)
            bomb = character:FindFirstChild(config.BombName) or backpack:FindFirstChild(config.BombName)
        end
        return bomb

        while true do
            task.wait(0.8)
            if visible then
                EnsureBombInBackpack()
            end

        cooldown = false

            label.Text = buttonDefaultText

                button.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)

    local function GetBomb()
        local character = LocalPlayer.Character
        local backpack = LocalPlayer:FindFirstChild('Backpack')
        if not character then return nil end

        local bomb = character:FindFirstChild(config.BombName)
        if bomb then return bomb end

        if backpack then
            bomb = backpack:FindFirstChild(config.BombName)
            if bomb then
                bomb.Parent = character
                return bomb
            end
        end

        bomb = EnsureBombInBackpack()
        if bomb and backpack and bomb.Parent == backpack then
            bomb.Parent = character
        end
        return bomb
    end

    local function DoJump()
        if cooldown then return end
        local character = LocalPlayer.Character
        if not character then return end

        local hrp = character:FindFirstChild('HumanoidRootPart')
        local humanoid = character:FindFirstChild('Humanoid')
        if not hrp or not humanoid then return end

        local bomb = GetBomb()
        if not bomb then return end

        local remote = bomb:FindFirstChild('Remote')
        if not remote then return end

        local camera = workspace.CurrentCamera
        local lookVector = camera.CFrame.LookVector
        local position = hrp.Position + (lookVector * 5) + Vector3.new(0, 2, 0)

        pcall(function()
            remote:FireServer(CFrame.new(position), 50)
        end)

        task.delay(0.05, function()
            if hrp then
                hrp.AssemblyLinearVelocity = Vector3.new(hrp.AssemblyLinearVelocity.X, math.max(hrp.AssemblyLinearVelocity.Y, 50), hrp.AssemblyLinearVelocity.Z)
            end
            humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
        end)

        cooldown = true

        task.spawn(function()
            for i = cooldownTime, 1, -1 do
                if not cooldown then break end
                label.Text = tostring(i)
                task.wait(1)
            end

            if not editMode then
                label.Text = buttonDefaultText
            end
            cooldown = false
        end)

        task.delay(0.4, function()
            if bomb and bomb.Parent == character then
                bomb.Parent = LocalPlayer.Backpack
            end
        end)
    end

    button.MouseButton1Click:Connect(function()
        if editMode then return end
        PlayClickAnimation()
        DoJump()
    end)

    UIS.InputBegan:Connect(function(input, gameProcessed)
        if gameProcessed or editMode or not currentKeybind then return end
        if input.KeyCode == currentKeybind then
            PlayClickAnimation()
            DoJump()
        end
    end)

        Title = 'Show Button',
        Default = false,

            end
        end,
    })

        Default = false,
        Callback = function(state)
            editMode = state
            if editMode then
                label.Text = 'MOVE'
            else
                if not cooldown then
                    label.Text = buttonDefaultText
                end
            end
        end,
    })

        Title = 'Button Color',
        Default = activeThemeColor,
        Callback = function(color)

        end,
    })

    config.Tab:Slider({
        Title = 'Button Size',
        Value = { Min = 30, Max = 200, Default = size },
        Callback = function(v)

        end,
    })

    config.Tab:Keybind({
        Title = 'Jump Keybind',
        Default = currentKeybind,
        Callback = function(key)
            currentKeybind = key
        end,
    })
end

local PlayerData = {}
pcall(function()
    local remotes = ReplicatedStorage:WaitForChild("Remotes", 3)
    if remotes then
        local gameplay = remotes:FindFirstChild("Gameplay", true)
        if gameplay then
            local pdEvent = gameplay:FindFirstChild("PlayerDataChanged")
            if pdEvent and pdEvent:IsA("RemoteEvent") then
                pdEvent.OnClientEvent:Connect(function(data)
                    if type(data) == "table" then
                        PlayerData = data
                    end
                end)
            end
        end
    end
end)

local function FindMurderer()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr ~= LocalPlayer then
            if (plr:FindFirstChild("Backpack") and plr.Backpack:FindFirstChild("Knife")) or 
               (plr.Character and plr.Character:FindFirstChild("Knife")) then
                return plr
            end
        end
    end
    for playerName, data in pairs(PlayerData) do
        if data and (data.Role == "Murderer" or data.role == "Murderer") then
            local found = Players:FindFirstChild(playerName)
            if found and found ~= LocalPlayer then return found end
        end
    end
    return nil
end

local function GetPredictedPosition(player)
    if not player or not player.Character then return Vector3.new(0,0,0) end
    local hrp = player.Character:FindFirstChild("HumanoidRootPart") or player.Character.PrimaryPart
    if not hrp then return Vector3.new(0,0,0) end

    local vel = hrp.AssemblyLinearVelocity or hrp.Velocity or Vector3.new(0,0,0)
    local ping = 0
    pcall(function()
        if LocalPlayer and typeof(LocalPlayer.GetNetworkPing) == "function" then
            ping = LocalPlayer:GetNetworkPing() or 0
        end
    end)
    if type(ping) ~= "number" then ping = tonumber(ping) or 0 end
    if ping > 1 then ping = ping / 1000 end
    local offset = 2.8
    local leadTime = math.max(0, offset * 0.01) + ping

    return hrp.Position + vel * leadTime
end

local function ShootMurdererOnce()
    local character = LocalPlayer.Character
    if not character then return end

    local target = FindMurderer()
    if not target or not target.Character then return end
    
    local mHRP = target.Character:FindFirstChild("HumanoidRootPart")
    local lHRP = character:FindFirstChild("HumanoidRootPart")
    if not mHRP or not lHRP then return end

    local rayParams = RaycastParams.new()
    rayParams.FilterType = Enum.RaycastFilterType.Exclude
    rayParams.FilterDescendantsInstances = { character }
    local direction = (mHRP.Position - lHRP.Position)
    local ok, res = pcall(function()
        return workspace:Raycast(lHRP.Position, direction, rayParams)
    end)

    local clearShot = false
    if not ok or not res then
        clearShot = true
    else
        if res.Instance and target.Character and res.Instance:IsDescendantOf(target.Character) then
            clearShot = true
        end
    end

    if not clearShot then return end

    if not character:FindFirstChild("Gun") then
        local tool = LocalPlayer.Backpack and LocalPlayer.Backpack:FindFirstChild("Gun")
        local humanoid = character:FindFirstChildOfClass("Humanoid")
        if tool and humanoid then
            pcall(function() humanoid:EquipTool(tool) end)
            task.wait(0.08)
        end
    end

    local gun = character:FindFirstChild("Gun")
    if not gun then return end

    local knifeLocal = gun:FindFirstChild("KnifeLocal")
    local createBeam = knifeLocal and knifeLocal:FindFirstChild("CreateBeam")
    local rf = createBeam and createBeam:FindFirstChild("RemoteFunction")
    
    if gun and knifeLocal and createBeam and rf and typeof(rf.InvokeServer) == "function" then
        local predPos = GetPredictedPosition(target)
        pcall(function()
            rf:InvokeServer(1, predPos, "AH2")
        end)
    end
end

local espGui = Instance.new("Folder")

local function GetPlayerRole(player)
    if not player or not player.Character then return "Innocent" end
    local backpack = player:FindFirstChild("Backpack")
    local character = player.Character

    local hasKnife = (backpack and backpack:FindFirstChild("Knife")) or character:FindFirstChild("Knife")
    local hasGun = (backpack and backpack:FindFirstChild("Gun")) or character:FindFirstChild("Gun")

    if hasKnife then return "Murderer"
    elseif hasGun then return "Sheriff" end
    return "Innocent"
end

local function GetRoleColor(role)
    if role == "Murderer" then return ESP_Config.MurdererColor
    elseif role == "Sheriff" then return ESP_Config.SheriffColor
    elseif role == "Hero" then return ESP_Config.HeroColor
    else return ESP_Config.InnocentColor end
end

local espObjects = {}

local function ClearESP(player)
    if espObjects[player] then
        for _, obj in pairs(espObjects[player]) do
            if obj then pcall(function() obj:Destroy() end) end
        end
        espObjects[player] = nil
    end
end

            gunDropHighlight.OutlineColor = Color3.fromRGB(255, 255, 255)

            lbl.BackgroundTransparency = 1
            lbl.Font = Enum.Font.GothamBold

            lbl.TextColor3 = ESP_Config.GunDropColor

        gunDropBillboard.Enabled = true
    else
        if gunDropHighlight then gunDropHighlight.Enabled = false end
        if gunDropBillboard then gunDropBillboard.Enabled = false end
    end
end

RunService.RenderStepped:Connect(function()
    UpdateGunDropESP()

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer then
            local data = espObjects[player]
            if not data then
                CreateESPForPlayer(player)
                data = espObjects[player]
            end

            if data and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
                local char = player.Character
                local hrp = char.HumanoidRootPart
                local role = GetPlayerRole(player)
                local color = GetRoleColor(role)

                if ESP_Config.Enabled then
                    if ESP_Config.Boxes or ESP_Config.Roles then
                        data.Highlight.Adornee = char
                        data.Highlight.FillColor = color
                        data.Highlight.OutlineColor = color
                        data.Highlight.Enabled = true
                    else
                        data.Highlight.Enabled = false
                    end

                    if ESP_Config.Names or ESP_Config.Roles then
                        data.Billboard.Adornee = hrp
                        local text = ""
                        if ESP_Config.Names then text = text .. player.DisplayName end
                        if ESP_Config.Roles then text = text .. " [" .. role .. "]" end
                        data.Label.Text = text
                        data.Label.TextColor3 = color
                        data.Billboard.Enabled = true
                    else
                        data.Billboard.Enabled = false
                    end
                else
                    data.Highlight.Enabled = false
                    data.Billboard.Enabled = false
                end
            elseif data then
                data.Highlight.Enabled = false
                data.Billboard.Enabled = false
            end
        end
    end
end)
